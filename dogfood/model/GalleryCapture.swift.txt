// Throwaway: renders a board file's boards to PNGs, for looking at a drawing made with canvas draw.
// GALLERY_FILE is the board file, GALLERY_OUT the folder, GALLERY_BOARDS the boards (titles of their
// owners, comma-separated; "root" for the top one; all when unset), GALLERY_ZOOM the zoom (0.5).
@MainActor let galleryTests = Suite("Gallery", [
    test("renders a board file's boards") {
        let env = ProcessInfo.processInfo.environment
        guard let path = env["GALLERY_FILE"], let out = env["GALLERY_OUT"] else { return }
        try configureBoards(pasteboard: NSPasteboard.withUniqueName())
        let board = BoardDocument()
        defer { board.close() }
        try board.read(from: Data(contentsOf: URL(fileURLWithPath: path)), ofType: boardType)
        let editor = board.editor
        let wanted = (env["GALLERY_BOARDS"] ?? "").split(separator: ",").map(String.init)
        let zoom = CGFloat(Double(env["GALLERY_ZOOM"] ?? "") ?? 0.5)
        @MainActor func render(_ name: String) throws {
            guard let bounds = editor.shapeContext.bounds(of: editor.document.shapes.map(\.id)) else { return }
            let margin: CGFloat = 40
            var z = zoom
            let longest = max(bounds.width, bounds.height) * z + 2 * margin
            if longest > 9000 { z *= 9000 / longest }
            let size = CGSize(width: (bounds.width * z + 2 * margin).rounded(.up), height: (bounds.height * z + 2 * margin).rounded(.up))
            editor.camera = Camera(origin: CGPoint(x: bounds.minX - margin / z, y: bounds.minY - margin / z), zoom: z)
            let rep = renderBitmap(editor, size: size)
            try rep.representation(using: .png, properties: [:])!.write(to: URL(fileURLWithPath: out).appendingPathComponent(name + ".png"))
            print("rendered \(name): \(Int(size.width))x\(Int(size.height)) at \(Int(z * 100))%, \(editor.document.shapes.count) shapes")
        }
        if let dump = env["GALLERY_DUMP"] {
            // Where every board's shapes are: the board (its owner's title, or "root"), kind, id, title,
            // bounds, and for a pill the title of what it stands for; one JSON object a line.
            var lines: [String] = []
            func esc(_ s: String) -> String {
                s.replacingOccurrences(of: "\\", with: "\\\\").replacingOccurrences(of: "\"", with: "\\\"").replacingOccurrences(of: "\n", with: "\\n")
            }
            @MainActor func dumpShown(_ board: String) {
                for record in editor.document.shapes {
                    guard let b = editor.shapeContext.bounds(of: [record.id]) else { continue }
                    let title = editor.registry.title(of: record) ?? ""
                    var stands = ""
                    if let target = record.typed(as: Reference.self)?.props.target?.id, let original = editor.file.record(target) {
                        stands = editor.registry.title(of: original) ?? ""
                    }
                    lines.append("{\"board\": \"\(esc(board))\", \"id\": \"\(record.id)\", \"kind\": \"\(record.kind)\", \"title\": \"\(esc(title))\", \"for\": \"\(esc(stands))\", \"x\": \(b.minX), \"y\": \(b.minY), \"w\": \(b.width), \"h\": \(b.height)}")
                }
            }
            dumpShown("root")
            for (owner, _) in editor.file.children {
                guard let record = editor.file.record(owner), let title = editor.registry.title(of: record) else { continue }
                guard editor.showBoard(BoardID(owner: owner)) else { print("can't show \(title)"); continue }
                dumpShown(title)
            }
            _ = editor.showBoard(.root)
            try lines.joined(separator: "\n").write(toFile: dump, atomically: true, encoding: .utf8)
            print("dumped \(lines.count) shapes")
        }
        if wanted.isEmpty || wanted.contains("root") { try render("root") }
        for (owner, _) in editor.file.children {
            guard let record = editor.file.record(owner), let title = editor.registry.title(of: record) else { continue }
            guard wanted.isEmpty || wanted.contains(title) else { continue }
            guard editor.showBoard(BoardID(owner: owner)) else { print("can't show \(title)"); continue }
            try render(title.replacingOccurrences(of: "/", with: "_"))
        }
    },
])
